% ===================== HERO =====================
\begin{columns}[t,onlytextwidth]
  \begin{column}{0.40\paperwidth}
    \begin{block}{The big idea}
      \Large Distance is a \textbf{tunable resolution knob} on musical structure ---
      sweeping \done${}\to{}$\dthree${}\to{}$\dtwo\ distills a piece down to its
      \textbf{core cyclic skeleton}.\par\vskip0.9ex
      \large
      \begin{itemize}\setlength{\itemsep}{1.0ex}
        \item \textbf{Three path distances}: \done\ (min edges),
              \dtwo\ (min weight --- the only metric), \dthree\ (hybrid).
        \item \textbf{Proven ordering} $\textcolor{dcD2}{d_2}\!\le\!\textcolor{dcD3}{d_3}\!\le\!\textcolor{dcD1}{d_1}$
              lifts to a \textbf{nontrivial barcode injection}
              $\mathrm{bcd}_1(\textcolor{dcD2}{d_2})\!\hookrightarrow\!\mathrm{bcd}_1(\textcolor{dcD3}{d_3})\!\hookrightarrow\!\mathrm{bcd}_1(\textcolor{dcD1}{d_1})$
              (Heo--Choi--Jung 2025).
        \item \textbf{Validated} on real Korean court music --- an exact match to the paper's Table~1.
      \end{itemize}
    \end{block}
  \end{column}
  \begin{column}{0.59\paperwidth}
    \begin{block}{Cycle skeleton extraction}
      {\centering\includegraphics[width=\linewidth]{signature.pdf}\par}\vskip0.4ex
      {\Large On \emph{Sanghyeondodeuri} (geomungo): as \done${}\to{}$\dthree${}\to{}$\dtwo,
      the $H_1$ barcode thins $|\mathrm{bcd}_1|\,{=}\,\mathbf{7}\,{\to}\,\mathbf{4}\,{\to}\,\mathbf{3}$ --- only the robust core loops survive.}
    \end{block}

% ===================== 3 CONTENT COLUMNS =====================
\begin{columns}[t,onlytextwidth]
  % ---- COLUMN 1 ----
  \begin{column}{0.32\paperwidth}
    \begin{block}{From music to a graph}
      \large Each \textbf{node} $=$ (pitch, duration); an \textbf{edge} joins consecutive notes;
      the \textbf{weight} is $1/(\text{co-occurrence count})$ --- frequent neighbors are closer.
      Rests are skipped.
      \vskip0.4ex\centering
      \includegraphics[width=0.54\linewidth]{fig_network.pdf}\\[0.2ex]
      {\normalsize The music network of \emph{Sanghyeondodeuri} (geomungo).}
    \end{block}
    \begin{block}{Three path distances}
      \large Path $P$: \textbf{hops} $\ell(P)$, \textbf{weight} $\omega(P)=\sum_{e\in P}W(e)$; $h=\min_P\ell(P)$.
      \begin{itemize}\setlength{\itemsep}{0.35ex}
        \item[\done] $\textcolor{dcD1}{d_1}=\omega(P),\ \ell(P)=h$ --- a \textbf{fewest-edge} path.
        \item[\dtwo] $\textcolor{dcD2}{d_2}=\min_P\omega(P)$ --- least weight; \emph{the only metric}.
        \item[\dthree] $\textcolor{dcD3}{d_3}=\min\{\omega(P):\ell(P)=h\}$ --- hybrid (least weight, fewest edges).
      \end{itemize}
      \vskip0.5ex\centering
      \begin{tikzpicture}[scale=1.5,every node/.style={circle,draw=dcInk,fill=dcPaper,
          line width=1pt,inner sep=2pt,font=\normalsize}]
        \node (s) at (0,0) {$s$};   \node (u) at (2,2.0) {$u$};   \node (v) at (2,0) {$v$};
        \node (w) at (1.1,-2.0) {$w$};   \node (x) at (2.9,-2.0) {$x$};   \node (t) at (4,0) {$t$};
        \draw[dcD1,line width=3pt] (s)--(u) node[midway,above left=-2pt,draw=none,fill=none,font=\small]{2};
        \draw[dcD1,line width=3pt] (u)--(t) node[midway,above right=-2pt,draw=none,fill=none,font=\small]{2};
        \draw[dcD3,line width=3pt] (s)--(v) node[midway,above=1pt,draw=none,fill=none,font=\small]{1};
        \draw[dcD3,line width=3pt] (v)--(t) node[midway,above=1pt,draw=none,fill=none,font=\small]{1};
        \draw[dcD2,line width=3pt] (s)--(w) node[midway,below left=-2pt,draw=none,fill=none,font=\small]{$\tfrac12$};
        \draw[dcD2,line width=3pt] (w)--(x) node[midway,below=1pt,draw=none,fill=none,font=\small]{$\tfrac12$};
        \draw[dcD2,line width=3pt] (x)--(t) node[midway,below right=-2pt,draw=none,fill=none,font=\small]{$\tfrac12$};
      \end{tikzpicture}\\[0.3ex]
      {\normalsize $s\!\to\!t$:\; $\textcolor{dcD1}{d_1{=}4}>\textcolor{dcD3}{d_3{=}2}>\textcolor{dcD2}{d_2{=}1.5}$.
       Both $s\,u\,t$ and $s\,v\,t$ are fewest-edge; $\textcolor{dcD1}{d_1}$ follows the one picked by
       \textbf{vertex order} ($u$ before $v$), $\textcolor{dcD3}{d_3}$ the lightest, $\textcolor{dcD2}{d_2}$ the lightest overall.}
    \end{block}
    \begin{block}{Only \dtwo\ is a metric}
      \large $\textcolor{dcD2}{d_2}$ satisfies the triangle inequality;
      $\textcolor{dcD1}{d_1}$ and $\textcolor{dcD3}{d_3}$ can violate it
      (a heavy direct edge). Yet all three give meaningful musical structure.
    \end{block}
  \end{column}
  % ---- COLUMN 2 ----
  \begin{column}{0.32\paperwidth}
    \begin{block}{A universal ordering}
      \large For \emph{every} pair of nodes,
      \[ \textcolor{dcD2}{d_2(v,w)} \;\le\; \textcolor{dcD3}{d_3(v,w)} \;\le\; \textcolor{dcD1}{d_1(v,w)}. \]
      This carries over to one-dimensional persistent homology.
    \end{block}
    \begin{block}{Barcode injection}
      \large The birth edges nest, $B_2\subseteq B_3\subseteq B_1$, giving injections
      \[ \mathrm{bcd}_1(\textcolor{dcD2}{d_2}) \hookrightarrow
         \mathrm{bcd}_1(\textcolor{dcD3}{d_3}) \hookrightarrow
         \mathrm{bcd}_1(\textcolor{dcD1}{d_1}). \]
      A \textbf{nontrivial theorem} for path-representable distances (\textbf{Heo--Choi--Jung 2025}, arXiv:2501.03553).\par\vskip0.3ex
      Each cycle keeps its \textbf{birth}; its \textbf{lifespan grows}
      $\textcolor{dcD2}{d_2}\!\le\!\textcolor{dcD3}{d_3}\!\le\!\textcolor{dcD1}{d_1}$,
      and the number of cycles can only increase.
      \vskip0.5ex\centering
      \begin{tikzpicture}[xscale=2.6,yscale=1.3]
        \foreach \y/\bx/\dx/\col in {2/0/3.0/dcD1, 1/0/2.0/dcD3, 0/0/1.2/dcD2}{
          \draw[\col,line width=6pt,line cap=round] (\bx,\y) -- (\dx,\y);
        }
        \node[right,font=\large] at (3.1,2) {\textcolor{dcD1}{$d_1$}: longest};
        \node[right,font=\large] at (2.1,1) {\textcolor{dcD3}{$d_3$}};
        \node[right,font=\large] at (1.3,0) {\textcolor{dcD2}{$d_2$}: shortest};
        \draw[dcInk,dashed,line width=1pt] (0,-0.45) -- (0,2.55) node[above,font=\normalsize]{shared birth};
      \end{tikzpicture}\\[0.4ex]
      \includegraphics[width=0.62\linewidth]{fig_persistence_diagram.pdf}\\[0.2ex]
      {\normalsize On real music: \textcolor{dcD1}{$d_1$} points lie above
       \textcolor{dcD2}{$d_2$} --- longer lifespans, shared births.}
    \end{block}
    \begin{block}{Why it matters}
      \large The same piece yields \emph{different but nested} topological readings.
      $\textcolor{dcD2}{d_2}$ exposes the robust core; $\textcolor{dcD1}{d_1}$ adds finer,
      peripheral loops. The changes are \textbf{systematic, not random}.
    \end{block}
  \end{column}
  % ---- COLUMN 3 ----
  \begin{column}{0.32\paperwidth}
    \begin{block}{Evidence on real music}
      \centering
      \includegraphics[width=\linewidth]{fig_distance_matrices.pdf}\\[0.3ex]
      {\normalsize Distance matrices: \textcolor{dcD2}{$d_2$} (cool/smooth) $\to$
       \textcolor{dcD1}{$d_1$} (hot)\\ the ordering is visible.}\\[0.9ex]
      \includegraphics[width=\linewidth]{fig_barcodes.pdf}\\[0.3ex]
      {\normalsize $H_1$ barcodes: $|\mathrm{bcd}_1(\textcolor{dcD1}{d_1})|,\,|\mathrm{bcd}_1(\textcolor{dcD3}{d_3})|,\,|\mathrm{bcd}_1(\textcolor{dcD2}{d_2})| = \mathbf{7,4,3}$;
       lifespans grow, births shared.}
    \end{block}
    \begin{block}{Distilling the core}
      \large Ranking $\textcolor{dcD1}{d_1}$ cycles by how far down they survive
      ($\to\textcolor{dcD3}{d_3}\to\textcolor{dcD2}{d_2}$) gives a principled
      \textbf{importance hierarchy of musical loops} --- a multi-resolution descriptor of a piece.
    \end{block}
    \begin{block}{Takeaways \& future work}
      \large
      \begin{itemize}\setlength{\itemsep}{0.6ex}
        \item Distance choice $=$ a tunable lens; results are nested, not arbitrary.
        \item Toward \textbf{AI music composition} with multiple stylistic perspectives
              from one source.
      \end{itemize}
